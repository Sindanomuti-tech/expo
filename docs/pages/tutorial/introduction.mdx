import React, { useState } from 'react';
import { StyleSheet, Text, View, TouchableOpacity, ScrollView, SafeAreaView, Alert } from 'react-native';

export default function App() {
  // Lista de produtos de exemplo para o PDV (Cantina / Retalho)
  const [produtos] = useState([
    { id: '1', nome: 'Cerveja Cuca 33cl', preco: 450 },
    { id: '2', nome: 'Água Pura 1.5L', preco: 250 },
    { id: '3', nome: 'Pacote de Bolacha', preco: 300 },
    { id: '4', nome: 'Refrigerante 33cl', preco: 400 },
  ]);

  // Carrinho de compras
  const [carrinho, setCarrinho] = useState([]);

  // Adicionar produto ao carrinho
  const adicionarAoCarrinho = (produto) => {
    setCarrinho([...carrinho, produto]);
  };

  // Calcular o total da fatura
  const calcularTotal = () => {
    return carrinho.reduce((soma, item) => soma + item.preco, 0);
  };

  // Emitir Fatura / Simulação AGT
  const emitirFatura = () => {
    if (carrinho.length === 0) {
      Alert.alert("Atenção", "O carrinho está vazio!");
      return;
    }
    const total = calcularTotal();
    Alert.alert(
      "Fatura Emitida com Sucesso!",
      `Total: ${total} Kz\nHash AGT: [OK-VERIFICADO]\nCódigo QR gerado para o cliente.`
    );
    setCarrinho([]); // Limpa o carrinho após emissão
  };

  return (
    <SafeAreaView style={styles.container}>
      {/* Cabeçalho com identidade visual AGT Pagamento */}
      <View style={styles.header}>
        <Text style={styles.headerTitle}>AGT Pagamento</Text>
        <Text style={styles.headerSub}>Sistema de Faturação Certificado</Text>
      </View>

      {/* Secção de Produtos (Botões de toque rápido) */}
      <View style={styles.section}>
        <Text style={styles.sectionTitle}>Toque para adicionar ao cesto:</Text>
        <ScrollView horizontal showsHorizontalScrollIndicator={false} style={styles.produtosScroll}>
          {produtos.map((p) => (
            <TouchableOpacity 
              key={p.id} 
              style={styles.produtoCard}
              onPress={() => adicionarAoCarrinho(p)}
            >
              <Text style={styles.produtoNome}>{p.nome}</Text>
              <Text style={styles.produtoPreco}>{p.preco} Kz</Text>
            </TouchableOpacity>
          ))}
        </ScrollView>
      </View>

      {/* Resumo do Carrinho / Venda Atual */}
      <View style={styles.carrinhoContainer}>
        <Text style={styles.sectionTitle}>Itens na Fatura Atual:</Text>
        <ScrollView style={styles.carrinhoLista}>
          {carrinho.length === 0 ? (
            <Text style={styles.carrinhoVazio}>Nenhum produto selecionado.</Text>
          ) : (
            carrinho.map((item, index) => (
              <View key={index} style={styles.carrinhoItem}>
                <Text style={styles.itemTexto}>{item.nome}</Text>
                <Text style={styles.itemTexto}>{item.preco} Kz</Text>
              </View>
            ))
          )}
        </ScrollView>
      </View>

      {/* Rodapé com Total e Botão de Emissão */}
      <View style={styles.footer}>
        <View style={styles.totalRow}>
          <Text style={styles.totalLabel}>Total a Pagar:</Text>
          <Text style={styles.totalValor}>{calcularTotal()} Kz</Text>
        </View>

        <TouchableOpacity style={styles.botaoEmitir} onPress={emitirFatura}>
          <Text style={styles.botaoEmitirTexto}>EMITIR FATURA AGT</Text>
        </TouchableOpacity>
      </View>
    </SafeAreaView>
  );
}
